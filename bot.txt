# 📩 Безопасная пересылка ответа админа клиенту
@dp.message(F.from_user.id == ADMIN_ID, F.reply_to_message)
async def reply_to_user(message: types.Message):
    global last_user_id
    
    # 1.Попробуем забрать ID из оригинального сообщения
    user_id = None
    if message.reply_to_message.forward_from:
        user_id = message.reply_to_message.forward_from.id
    elif last_user_id:
        user_id = last_user_id
        
    if user_id:
        try:
            await bot.send_message(chat_id=user_id, text=message.text)
            await message.answer("✅ Ваш ответ отправлен клиенту!")
        except Exception as e:
            await message.answer(f"❌ Ошибка отправки: {e}")
    else:
        await message.answer("⚠️ У пользователя закрытый аккаунт, и бот не смог определить его ID.")
